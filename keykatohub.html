-- ==================================================== -- TÊN FILE LƯU BỘ NHỚ TRÊN THIẾT BỊ NGƯỜI DÙNG -- ==================================================== local saveFileName = "KatoHub_LastKeyDate.txt" -- 1. Hàm tính Key chuẩn của NGÀY HÔM NAY (Múi giờ UTC) local function getTodayKey() local baseKey = "KatoHub-premeumprovip" local date = os.date("!*t") local dateSeed = date.year * 10000 + date.month * 100 + date.day local randomCode = math.floor(math.abs(math.sin(dateSeed) * 1000000) % 9000) + 1000 -- Trả về Key hôm nay & Mã ngày dạng YYYYMMDD return baseKey .. "-" .. tostring(randomCode), tostring(dateSeed) end -- 2. Chúc bạn chơi game vui vẻ.", Duration = 5 }) return true -- TRƯỜNG HỢP 2: Nhập lại Key cũ (Key hôm nay khác ngày đã lưu gần nhất) elseif lastSavedSeed ~= "" and lastSavedSeed ~= todaySeed then game:GetService("StarterGui"):SetCore("SendNotification", { Title = "script:KatoHub - Báo Lỗi", Text = "Key đã hết hạn! Vui lòng getkey của ngày hôm nay.", Duration = 6 }) return false -- TRƯỜNG HỢP 3: Nhập sai Key hoàn toàn else game:GetService("StarterGui"):SetCore("SendNotification", { Title = "script:KatoHub - Báo Lỗi", Text = "Key không chính xác! Vui lòng kiểm tra lại.", Duration = 5 }) return false end end -- ==================================================== -- VÍ DỤ CÁCH SỬ DỤNG KHI NGƯỜI DÙNG BẤM NÚT XÁC NHẬN KEY -- ==================================================== -- Thay `KeyDinhDan` bằng ô Textbox chứa Key của sếp local isSuccess = checkUserKey(KeyDinhDan) if isSuccess then -- ================================================ -- CODE CHÍNH CỦA SẾP ĐẶT Ở ĐÂY -- ================================================ print("Script KatoHub bắt đầu chạy...") end
